**Goal:** забирать в brain из Gmail и Яндекс.Почты по IMAP только мои переписки и помеченные треды за 2 года, с пересказом той же моделью.

```python
"""Пароли из связки ключей macOS.

Пароли приложений почты не храним в config.json: пользователь кладёт их в связку ключей
командой `security add-generic-password`, а мы читаем на каждый прогон и нигде не сохраняем.
"""
from __future__ import annotations

import subprocess
from typing import List, Optional

SECURITY = "/usr/bin/security"


def add_command(service: str, account: str) -> str:
    """Команда, которой пользователь кладёт пароль: она сама его спросит, в историю терминала он не попадёт."""
    return f"security add-generic-password -s {service} -a {account} -w"


def password(service: str, account: str, command: Optional[List[str]] = None) -> str:
    cmd = command or [SECURITY, "find-generic-password", "-s", service, "-a", account, "-w"]
    try:
        out = subprocess.run(cmd, capture_output=True, text=True, timeout=20)
    except (OSError, subprocess.TimeoutExpired) as exc:
        raise RuntimeError(f"не получилось прочитать связку ключей ({exc})") from None
    secret = out.stdout.rstrip("\n")
    if out.returncode != 0 or not secret:
        raise RuntimeError(f"нет пароля в связке ключей — выполни `{add_command(service, account)}`")
    return secret
```

```python
CARD_MASK = "[карта скрыта]"
# 13–19 цифр подряд или группами через пробел/дефис. Проверка Луна отсекает номера заказов и телефоны.
CARD = re.compile(r"(?<![\d-])(?:\d[ -]?){12,18}\d(?![\d-])")


def _luhn(digits: str) -> bool:
    total = 0
    for i, ch in enumerate(reversed(digits)):
        d = int(ch)
        if i % 2:
            d = d * 2 - 9 if d > 4 else d * 2
        total += d
    return total % 10 == 0


def _card(match: "re.Match") -> str:
    digits = re.sub(r"\D", "", match.group())
    return CARD_MASK if 13 <= len(digits) <= 19 and _luhn(digits) else match.group()
```

```python
def _drop_from_queue(conn: sqlite3.Connection, ids) -> int:
    """Убирает записи из очереди без карточки — они ищутся по своему тексту. Старую карточку (если запись
    раньше пересказывали) стираем; summarized_at не ставим — в «последние карточки» они не попадают."""
    for rid in ids:
        conn.execute(
            """UPDATE records SET ai_title = NULL, summary = NULL, questions = NULL, keywords = NULL,
                   atoms = NULL, summarized_hash = content_hash, summarized_at = NULL, summary_error = NULL
               WHERE id = ?""",
            (rid,),
        )
        db.reindex(conn, rid)
    conn.commit()
    return len(ids)


def skip_short(conn: sqlite3.Connection, min_chars: int) -> int:
    """Записи короче min_chars модели не отправляем: пересказывать нечего."""
    return _drop_from_queue(conn, [r["id"] for r in conn.execute(
        f"SELECT id FROM records WHERE {PENDING} AND length(body) < ?", (min_chars,))])

               if cfg["sources"].get(c.name, {}).get("summarize") is False]
    if not sources:
        return 0
    marks = ", ".join("?" * len(sources))
    return _drop_from_queue(conn, [r["id"] for r in conn.execute(
        f"SELECT id FROM records WHERE {PENDING} AND source IN ({marks})", sources)])
```

```python
def eml(msg_id, sender, to, subject, text=None, html=None, date="Thu, 01 Oct 2026 10:00:00 +0300",
        in_reply_to=None, references=(), attachments=()):
    """Письмо в байтах, как его отдаёт IMAP."""
    m = EmailMessage()
    m["Message-ID"] = f"<{msg_id}>"
    m["From"] = sender
    m["To"] = to
    m["Subject"] = subject
    m["Date"] = date
    if in_reply_to:
        m["In-Reply-To"] = f"<{in_reply_to}>"
    if references:
        m["References"] = " ".join(f"<{r}>" for r in references)
    if text is not None:
        m.set_content(text)
        if html is not None:
            m.add_alternative(html, subtype="html")
    elif html is not None:
        m.set_content(html, subtype="html")
    else:
        m.set_content("")
    for name in attachments:
        m.add_attachment(b"data", maintype="application", subtype="octet-stream", filename=name)
    return m.as_bytes()
```

```python
"""Разбор писем без сети: байты письма → MailMessage, HTML → текст, без цитат и подписи.

Чистые функции: их проверяют тесты на готовых письмах, а IMAP живёт отдельно в mailbox.py.
"""
from __future__ import annotations

import email
import email.policy
import re
from dataclasses import dataclass, field
from email.utils import getaddresses, parsedate_to_datetime
from html.parser import HTMLParser
from typing import List, Tuple

from ..timeutil import to_iso
from .base import clip

MESSAGE_ID = re.compile(r"<([^<>\s]+)>")
SUBJECT_PREFIX = re.compile(r"^\s*(?:(?:re|fwd?|aw|wg|ответ|отв|пересл)\s*(?:\[\d+\]|\(\d+\))?\s*:\s*)+", re.I)
# Шапка цитаты: ниже неё — прошлые письма, они в треде и так есть.
QUOTE_HEADER = re.compile(
    r"^\s*(?:"
    r"On\s.{0,300}\swrote:"                                                       # по-английски
    r"|.{0,300}(?:\d{1,2}:\d{2}|@).{0,300}(?:написал(?:\(а\)|а)?|пишет):"        # по-русски
    r"|.{0,200}\d{1,2}:\d{2}.{0,200}<[^<>\s@]+@[^<>\s]+>:"                        # «…, 10:00, Анна <a@x.ru>:»
    r"|-{2,}\s*(?:Original Message|Исходное сообщение)\s*-{2,}"
    r")\s*$", re.I)
OUTLOOK_FROM = re.compile(r"^\s*(?:From|От):\s", re.I)
OUTLOOK_SENT = re.compile(r"^\s*(?:Sent|Отправлено|Date|Дата):\s", re.I)
# Пересылаемое письмо — это содержание, после такой строки ничего не режем.
FORWARDED = re.compile(r"^\s*-{5,}\s*(?:Forwarded message|Пересылаемое сообщение)\s*-{5,}\s*$", re.I)
BLOCK_TAGS = {"br", "p", "div", "li", "tr", "h1", "h2", "h3", "h4", "h5", "h6", "table", "hr"}
SKIP_TAGS = {"script", "style", "head", "title", "blockquote"}  # blockquote — цитата прошлого письма


@dataclass
class MailMessage:
    message_id: str                    # без угловых скобок; у письма без заголовка — пусто
    in_reply_to: str
    references: List[str]
    date: str                          # UTC ISO
    sender: Tuple[str, str]            # (имя, адрес в нижнем регистре)
    recipients: List[Tuple[str, str]]
    subject: str
    text: str
    attachments: List[str] = field(default_factory=list)
    mine: bool = False                 # моё письмо — решает стратегия ящика
    flagged: bool = False


class _Text(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.parts: List[str] = []
        self.skip = 0

    def handle_starttag(self, tag, attrs):
        if tag in SKIP_TAGS:
            self.skip += 1
        elif tag in BLOCK_TAGS:
            self.parts.append("\n")

    def handle_endtag(self, tag):
        if tag in SKIP_TAGS:
            self.skip = max(0, self.skip - 1)
        elif tag in BLOCK_TAGS:
            self.parts.append("\n")

    def handle_data(self, data):
        if not self.skip:
            self.parts.append(data)


def html_to_text(markup: str) -> str:
    parser = _Text()
    parser.feed(markup)
    parser.close()
    lines = [" ".join(line.replace("\xa0", " ").split()) for line in "".join(parser.parts).splitlines()]
    return re.sub(r"\n{3,}", "\n\n", "\n".join(lines)).strip()


def strip_quotes(text: str) -> str:
    lines = text.replace("\r\n", "\n").split("\n")
    out: List[str] = []
    for i, line in enumerate(lines):
        if FORWARDED.match(line):
            out.extend(lines[i:])
            break
        if line.rstrip() == "--":
            break  # подпись
        if QUOTE_HEADER.match(line):
            break
        nxt = lines[i + 1] if i + 1 < len(lines) else ""
        # Длинную шапку «On … wrote:» почтовики переносят на вторую строку.
        if line.lstrip().startswith("On ") and QUOTE_HEADER.match(line + " " + nxt):
            break
        if OUTLOOK_FROM.match(line) and OUTLOOK_SENT.match(nxt):
            break
        if line.lstrip().startswith(">"):
            continue
        out.append(line)
    return re.sub(r"\n{3,}", "\n\n", "\n".join(out)).strip()


def normalize_subject(subject: str) -> str:
    return SUBJECT_PREFIX.sub("", subject or "").strip() or "(без темы)"


def thread_root(message: MailMessage) -> str:
    """Корень треда: первый из References, иначе In-Reply-To, иначе само письмо."""
    if message.references:
        return message.references[0]
    return message.in_reply_to or message.message_id


def _header(msg, name: str) -> str:
    try:
        return str(msg.get(name) or "").strip()
    except Exception:  # кривой заголовок — не повод терять письмо
        return ""


def _addresses(value: str) -> List[Tuple[str, str]]:
    return [(n.strip(), a.strip().lower()) for n, a in getaddresses([value]) if a.strip()]


def _date(value: str, fallback: str) -> str:
    try:
        return to_iso(parsedate_to_datetime(value).timestamp()) or fallback
    except (TypeError, ValueError, IndexError, OverflowError):
        return fallback


def _content(part) -> str:
    try:
        return part.get_content()
    except (LookupError, UnicodeError, KeyError, AssertionError):  # неизвестная кодировка
        payload = part.get_payload(decode=True) or b""
        return payload.decode("utf-8", errors="replace")


def _body(msg) -> Tuple[str, List[str]]:
    names = [str(part.get_filename()) for part in msg.walk() if part.get_filename()]
    part = msg.get_body(preferencelist=("plain", "html"))
    if part is None:
        return "", names
    text = _content(part)
    if part.get_content_subtype() == "html":
        text = html_to_text(text)
    return strip_quotes(text), names


def parse(raw: bytes, fallback_date: str = "") -> MailMessage:
    try:
        msg = email.message_from_bytes(raw, policy=email.policy.default)
    except Exception as exc:
        return MailMessage("", "", [], fallback_date, ("", ""), [], "", f"[не удалось разобрать письмо: {exc}]")
    ids = MESSAGE_ID.findall(_header(msg, "Message-ID"))
    reply = MESSAGE_ID.findall(_header(msg, "In-Reply-To"))
    senders = _addresses(_header(msg, "From"))
    try:
        text, names = _body(msg)
    except Exception as exc:  # битое письмо не должно стопорить источник
        text, names = f"[не удалось разобрать письмо: {exc}]", []
    return MailMessage(
        message_id=ids[0] if ids else "",
        in_reply_to=reply[0] if reply else "",
        references=MESSAGE_ID.findall(_header(msg, "References")),
        date=_date(_header(msg, "Date"), fallback_date),
        sender=senders[0] if senders else ("", ""),
        recipients=_addresses(", ".join(_header(msg, h) for h in ("To", "Cc"))),
        subject=_header(msg, "Subject"),
        text=text,
        attachments=names,
    )


def _person(name: str, address: str) -> str:
    if name and address:
        return f"{name} <{address}>"
    return address or name or "(без отправителя)"


def render_thread(messages: List[MailMessage]) -> Tuple[str, str]:
    """Название и текст записи: шапка с участниками, затем письма по времени."""
    messages = sorted(messages, key=lambda m: m.date)
    title = normalize_subject(messages[0].subject)
    me = {m.sender[1] for m in messages if m.mine and m.sender[1]}
    people: List[str] = []
    seen = set()
    for m in messages:
        for name, address in [m.sender] + m.recipients:
            if address and address not in seen:
                seen.add(address)
                people.append(f"я <{address}>" if address in me else _person(name, address))
    blocks = []
    for m in messages:
        head = "я" if m.mine else _person(*m.sender)
        when = m.date[:16].replace("T", " ") if m.date else "—"
        block = f"{head}, {when}:\n{clip(m.text) if m.text else '(пустое письмо)'}"
        if m.attachments:
            block += "\n[вложения: " + ", ".join(m.attachments) + "]"
        blocks.append(block)
    head = "\n".join([f"Переписка: {title}", "Участники: " + ", ".join(people), f"Писем: {len(messages)}"])
    return title, head + "\n\n" + "\n\n".join(blocks)
```

```python
Логики «что брать» здесь нет — её держат стратегии Gmail и Яндекса (gmail.py, yandex_mail.py).
В тестах вместо Mailbox подставляется поддельный ящик с теми же методами.
"""
from __future__ import annotations

import imaplib
import re
import time
from dataclasses import dataclass, field
from datetime import datetime
from typing import Callable, Dict, Iterable, List, Optional, Tuple

from ..timeutil import to_iso
from .mailparse import MailMessage, parse

TIMEOUT = 60
FETCH_LIMIT = 1048576  # первый мегабайт письма: текст идёт раньше вложений, режутся в основном они
CHUNK = 200            # UID в одной команде FETCH
MONTHS_EN = ("Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec")
LIST_LINE = re.compile(rb'^\((?P<flags>[^)]*)\)\s+(?:"(?:[^"\\]|\\.)*"|NIL)\s+(?P<name>.+)$')
STATUS_VALUE = re.compile(rb"(UIDVALIDITY|UIDNEXT)\s+(\d+)")
UID = re.compile(rb"\bUID\s+(\d+)")
FLAGS = re.compile(rb"\bFLAGS\s+\(([^)]*)\)")
LABELS = re.compile(rb'\bX-GM-LABELS\s+\(((?:[^()"]|"(?:[^"\\]|\\.)*")*)\)')
THRID = re.compile(rb"\bX-GM-THRID\s+(\d+)")
INTERNALDATE = re.compile(rb'INTERNALDATE\s+"[^"]+"')
TOKEN = re.compile(rb'"((?:[^"\\]|\\.)*)"|([^\s()]+)')
NEW_ITEM = re.compile(rb"^\d+ \(")


class AuthFailed(RuntimeError):
    """Сервер не принял адрес и пароль."""


@dataclass
class Folder:
    name: str           # имя, как его понимает сервер (modified UTF-7) — для SELECT
    flags: List[str]    # \Sent, \All, \Junk…


@dataclass
class Fetched:
    uid: int
    raw: bytes = b""
    flags: List[str] = field(default_factory=list)
    labels: List[str] = field(default_factory=list)   # X-GM-LABELS — только Gmail
    thread: str = ""                                  # X-GM-THRID — только Gmail
    internal_date: str = ""                           # UTC ISO


def quote(value: str) -> str:
    return '"' + value.replace("\\", "\\\\").replace('"', '\\"') + '"'


def imap_date(day: datetime) -> str:
    """Дата для SEARCH SINCE: «07-Oct-2024» (английские месяцы, от локали не зависит)."""
    return f"{day.day:02d}-{MONTHS_EN[day.month - 1]}-{day.year}"


def gmail_date(day: datetime) -> str:
    return f"{day.year:04d}/{day.month:02d}/{day.day:02d}"


def _unescape(raw: bytes) -> str:
    return re.sub(rb"\\(.)", rb"\1", raw).decode("utf-8", errors="replace")


def _tokens(raw: bytes) -> List[str]:
    return [_unescape(quoted) if quoted else atom.decode("utf-8", errors="replace")
            for quoted, atom in TOKEN.findall(raw)]


def _name(raw: bytes) -> str:
    raw = raw.strip()
    if raw.startswith(b'"') and raw.endswith(b'"'):
        return _unescape(raw[1:-1])
    return raw.decode("utf-8", errors="replace")


def parse_list(lines) -> List[Folder]:
    folders = []
    for line in lines or []:
        if not isinstance(line, bytes):
            continue  # имя литералом — у наших почтовиков не встречается
        match = LIST_LINE.match(line)
        if match:
            folders.append(Folder(_name(match.group("name")), match.group("flags").decode().split()))
    return folders


def parse_status(data) -> Tuple[int, int]:
    values = dict(STATUS_VALUE.findall(b" ".join(d for d in data if isinstance(d, bytes))))
    return int(values.get(b"UIDVALIDITY", 0)), int(values.get(b"UIDNEXT", 1))


def _internal(raw: bytes) -> str:
    parsed = imaplib.Internaldate2tuple(raw)
    return to_iso(time.mktime(parsed)) if parsed else ""


def parse_fetch(data) -> List[Fetched]:
    """Ответ FETCH из imaplib: кортеж (шапка, литерал письма) и хвост шапки отдельной строкой,
    либо одна строка, если тело не просили."""
    items: List[Tuple[bytes, bytes]] = []
    for part in data or []:
        if isinstance(part, tuple):
            items.append((part[0], part[1]))
        elif isinstance(part, bytes):
            if NEW_ITEM.match(part) or not items:
                items.append((part, b""))
            else:
                meta, raw = items[-1]
                items[-1] = (meta + part, raw)
    out = []
    for meta, raw in items:
        uid = UID.search(meta)
        if not uid:
            continue
        flags, labels = FLAGS.search(meta), LABELS.search(meta)
        thread, idate = THRID.search(meta), INTERNALDATE.search(meta)
        out.append(Fetched(
            uid=int(uid.group(1)),
            raw=raw or b"",
            flags=_tokens(flags.group(1)) if flags else [],
            labels=_tokens(labels.group(1)) if labels else [],
            thread=thread.group(1).decode() if thread else "",
            internal_date=_internal(idate.group(0)) if idate else "",
        ))
    return out


class Mailbox:
    def __init__(self, host: str, address: str, password: str, gmail: bool = False,
                 connect: Optional[Callable[[], imaplib.IMAP4]] = None):
        self.address = address
        self.password = password
        self.gmail = gmail
        self._connect = connect or (lambda: imaplib.IMAP4_SSL(host, 993, timeout=TIMEOUT))
        self.imap = None

    def open(self) -> "Mailbox":
        self.imap = self._connect()
        try:
            self.imap.login(self.address, self.password)
        except imaplib.IMAP4.error as exc:
            raise AuthFailed(str(exc)) from None
        return self

    def close(self) -> None:
        try:
            self.imap.logout()
        except Exception:
            pass  # закрываем как можем: результат прогона от этого не зависит

    @staticmethod
    def _ok(typ, data, what: str):
        if typ != "OK":
            raise imaplib.IMAP4.error(f"{what}: {data!r}")
        return data

    def folders(self) -> List[Folder]:
        return parse_list(self._ok(*self.imap.list(), "LIST"))

    def select(self, name: str) -> Tuple[int, int]:
        """Открывает папку только на чтение (флаги «прочитано» не трогаем) и возвращает UIDVALIDITY и UIDNEXT."""
        validity, uidnext = parse_status(self._ok(*self.imap.status(quote(name), "(UIDVALIDITY UIDNEXT)"), "STATUS"))
        self._ok(*self.imap.select(quote(name), readonly=True), "SELECT")
        return validity, uidnext

    def _search(self, *criteria: str) -> List[int]:
        data = self._ok(*self.imap.uid("SEARCH", *criteria), "SEARCH")
        return sorted(int(x) for x in b" ".join(d for d in data if isinstance(d, bytes)).split())

    def uids_after(self, uid: int) -> List[int]:
        # «UID n:*» при n больше последнего UID отдаёт последнее письмо — его отсекаем.
        return [u for u in self._search("UID", f"{uid + 1}:*") if u > uid]

    def uids_since(self, day: datetime, flagged: bool = False) -> List[int]:
        criteria = ("FLAGGED", "SINCE", imap_date(day)) if flagged else ("SINCE", imap_date(day))
        return self._search(*criteria)

    def uids_in_thread(self, root: str) -> List[int]:
        q = quote(root)
        return self._search("OR", "OR", "HEADER", "Message-ID", q, "HEADER", "References", q,
                            "HEADER", "In-Reply-To", q)

    def gmail_search(self, query: str, thread: str = "") -> List[int]:
        criteria = ("X-GM-THRID", thread) if thread else ()
        return self._search(*criteria, "X-GM-RAW", quote(query))

    def gmail_thread_uids(self, thread: str) -> List[int]:
        return self._search("X-GM-THRID", thread)

    def _fetch(self, uids: Iterable[int], items: str) -> List[Fetched]:
        uids = sorted(set(uids))
        out: List[Fetched] = []
        for i in range(0, len(uids), CHUNK):
            chunk = ",".join(str(u) for u in uids[i:i + CHUNK])
            out.extend(parse_fetch(self._ok(*self.imap.uid("FETCH", chunk, items), "FETCH")))
        return out

    def gmail_threads(self, uids: Iterable[int]) -> Dict[int, str]:
        return {f.uid: f.thread for f in self._fetch(uids, "(UID X-GM-THRID)") if f.thread}

    def headers(self, uids: Iterable[int]) -> Dict[int, MailMessage]:
        items = "(UID BODY.PEEK[HEADER.FIELDS (MESSAGE-ID IN-REPLY-TO REFERENCES)])"
        return {f.uid: parse(f.raw) for f in self._fetch(uids, items)}

    def fetch(self, uids: Iterable[int], body: bool = True) -> List[Fetched]:
        items = ["UID", "FLAGS", "INTERNALDATE"]
        if self.gmail:
            items += ["X-GM-LABELS", "X-GM-THRID"]
        if body:
            items.append(f"BODY.PEEK[]<0.{FETCH_LIMIT}>")
        return self._fetch(uids, "(" + " ".join(items) + ")")
```

```python
"""Почта: общий алгоритм для IMAP-ящиков.

Что считать «моей перепиской» и как собрать тред, решают стратегии (gmail.py, yandex_mail.py);
здесь — пароль, вход, пачки, курсор и запись треда. Курсор нового прохода пишется только с последней
пачкой, промежуточные несут прежний: оборвался прогон — следующий повторит проход, уже записанные
треды отсечёт хеш, модель повторно не вызовется.
"""
from __future__ import annotations

import imaplib
import json
from datetime import datetime, timedelta, timezone
from typing import Iterator, List, Optional, Set, Tuple

from .. import keychain
from .base import ApiConnector, Batch, Record
from .mailbox import AuthFailed, Fetched, Mailbox
from .mailparse import MailMessage, parse, render_thread

THREAD_BATCH = 20
KEYCHAIN_SERVICE = "brain-mail"


class MailConnector(ApiConnector):
    default_every_minutes = 15
    empty_hint = ("Пока пусто. Включи ящик в config.json и положи пароль приложения в связку ключей "
                  "(README, раздел «Почта»).")
    host = ""
    gmail = False   # нужны ли расширения Gmail в FETCH
    auth_hint = ""  # что ещё проверить, если сервер не пускает

    def open_mailbox(self, address: str, secret: str):
        """Вход в ящик; тесты подменяют этот метод поддельным ящиком."""
        return Mailbox(self.settings.get("host") or self.host, address, secret, gmail=self.gmail).open()

    @property
    def since(self) -> datetime:
        return datetime.now(timezone.utc) - timedelta(days=self.backfill_days)

    def fetch(self, cursor: Optional[str]) -> Iterator[Batch]:
        address = self.settings.get("address") or ""
        if not address:
            raise RuntimeError(f"{self.title}: в config.json не задан address")
        try:
            secret = keychain.password(self.settings.get("keychain_service") or KEYCHAIN_SERVICE, address,
                                       self.settings.get("password_command"))
            box = self.open_mailbox(address, secret)
        except AuthFailed:
            raise RuntimeError(f"{self.title}: почта не пускает — проверь пароль приложения{self.auth_hint}") from None
        except RuntimeError as exc:
            raise RuntimeError(f"{self.title}: {exc}") from None
        except OSError as exc:
            raise RuntimeError(f"{self.title}: нет связи ({exc})") from None
        try:
            yield from self._pass(box, cursor)
        except OSError as exc:  # нет сети, таймаут, обрыв соединения
            raise RuntimeError(f"{self.title}: нет связи ({exc})") from None
        except imaplib.IMAP4.error as exc:
            raise RuntimeError(f"{self.title}: ошибка IMAP ({exc})") from None
        finally:
            box.close()

    def _pass(self, box, cursor: Optional[str]) -> Iterator[Batch]:
        previous = cursor or "{}"  # «{}» читается как «курсора нет»
        try:
            pos = json.loads(previous)
        except ValueError:
            pos = {}
        keys, after = self.candidates(box, pos if isinstance(pos, dict) else {})
        batch: List[Record] = []
        for key in sorted(keys):
            messages = self.thread(box, key)
            if not any(m.mine or m.flagged for m in messages):
                continue  # не моя переписка: ни моего письма, ни пометки
            batch.append(self.record(key, messages))
            if len(batch) >= THREAD_BATCH:
                yield Batch(batch, previous)
                batch = []
        yield Batch(batch, json.dumps(after, sort_keys=True))

    def record(self, key: str, messages: List[MailMessage]) -> Record:
        title, body = render_thread(messages)
        dates = sorted(m.date for m in messages if m.date)
        return Record(
            source=self.source,
            external_id=self.external_id(key),
            title=title,
            body=body,
            updated_at=dates[-1] if dates else "1970-01-01T00:00:00Z",
            started_at=dates[0] if dates else None,
            location=self.location(key),
            kind="email",
        )

    @staticmethod
    def message(fetched: Fetched, mine: bool) -> MailMessage:
        m = parse(fetched.raw, fetched.internal_date)
        m.mine = mine
        m.flagged = "\\Flagged" in fetched.flags
        return m

    # Стратегия ящика.

    def candidates(self, box, pos: dict) -> Tuple[Set[str], dict]:
        """Треды-кандидаты этого прохода и курсор, который запишется после него."""
        raise NotImplementedError

    def thread(self, box, key: str) -> List[MailMessage]:
        raise NotImplementedError

    def external_id(self, key: str) -> str:
        raise NotImplementedError

    def location(self, key: str) -> Optional[str]:
        return None
```

```python
    title = "Gmail"
    about = "переписка в Gmail"
    host = "imap.gmail.com"
    gmail = True

    def candidates(self, box, pos: dict) -> Tuple[Set[str], dict]:
        folder = next((f for f in box.folders() if "\\All" in f.flags), None)
        if folder is None:
            raise RuntimeError("Gmail: не нашлась папка «Вся почта» (флаг \\All)")
        validity, uidnext = box.select(folder.name)
        after = gmail_date(self.since)
        starred = set(box.gmail_threads(box.gmail_search(f"is:starred after:{after}")).values())
        if pos.get("validity") != validity or "uid" not in pos:
            # Первый проход: Gmail сам находит мои и помеченные письма — весь ящик читать не нужно.
            keys = set(box.gmail_threads(box.gmail_search(f"(in:sent OR is:starred) after:{after}")).values())
        else:
            # Новые письма пересобирают свой тред, только если он мой; плюс новые звёздочки на старых письмах.
            touched = set(box.gmail_threads(box.uids_after(int(pos["uid"]))).values())
            keys = {t for t in touched if box.gmail_search("in:sent OR is:starred", thread=t)}
            keys |= starred - set(pos.get("starred") or [])
        return keys, {"validity": validity, "uid": uidnext - 1, "starred": sorted(starred)}

    def thread(self, box, key: str) -> List[MailMessage]:
        return [self.message(f, "\\Sent" in f.labels) for f in box.fetch(box.gmail_thread_uids(key))]

    def external_id(self, key: str) -> str:
        return f"thr:{key}"

    def location(self, key: str) -> Optional[str]:
        return f"https://mail.google.com/mail/#all/{int(key):x}"  # так Gmail адресует тред в вебе
```

```python
"""Яндекс.Почта: треды по заголовкам Message-ID / In-Reply-To / References.

Моё письмо — то, что лежит в папке отправленных (флаг \\Sent), помеченное — с флагом \\Flagged.
Спам, корзину и черновики не читаем. Корень треда — первый из References, иначе In-Reply-To,
иначе само письмо; почтовик, который пишет только In-Reply-To, рвёт длинную переписку на части.
"""
from __future__ import annotations

from typing import Dict, List, Set, Tuple

from .mail import MailConnector
from .mailparse import MailMessage, thread_root

SKIP_FLAGS = {"\\Junk", "\\Spam", "\\Trash", "\\Drafts", "\\Noselect"}


class YandexMailConnector(MailConnector):
    name = "yandex_mail"
    source = "yandex_mail"
    title = "Яндекс.Почта"
    about = "переписка в Яндекс.Почте"
    host = "imap.yandex.ru"
    auth_hint = " и включён ли IMAP в настройках почты («Почтовые программы»)"

    def candidates(self, box, pos: dict) -> Tuple[Set[str], dict]:
        self.folders = [f for f in box.folders() if not SKIP_FLAGS & set(f.flags)]
        self.sent = {f.name for f in self.folders if "\\Sent" in f.flags}
        known = pos.get("folders") or {}
        known_flagged = set(pos.get("flagged") or [])
        keys: Set[str] = set()
        flagged_now: Set[str] = set()
        after: Dict[str, dict] = {}
        for folder in self.folders:
            validity, uidnext = box.select(folder.name)
            old = known.get(folder.name) or {}
            fresh = old.get("validity") != validity or "uid" not in old
            sent = folder.name in self.sent
            if fresh:
                # Первый проход папки: мои треды начинаются с моих писем, чужие ответы найдутся при сборке.
                new = set(box.uids_since(self.since)) if sent else set()
            else:
                new = set(box.uids_after(int(old["uid"])))
            flagged = set(box.uids_since(self.since, flagged=True))
            for uid, head in box.headers(new | flagged).items():
                root = thread_root(head)
                if not root:
                    continue
                if uid in flagged:
                    flagged_now.add(root)
                    if root not in known_flagged:
                        keys.add(root)
                # Моё новое письмо или чей-то ответ — тред проверим целиком. Письмо без In-Reply-To и
                # References — начало чужого треда, моей переписки там ещё нет: новые рассылки ничего не стоят.
                if uid in new and (sent or head.in_reply_to or head.references):
                    keys.add(root)
            after[folder.name] = {"validity": validity, "uid": uidnext - 1}
        return keys, {"folders": after, "flagged": sorted(flagged_now)}

    def thread(self, box, root: str) -> List[MailMessage]:
        found: Dict[str, List[int]] = {}
        for folder in self.folders:
            box.select(folder.name)
            uids = box.uids_in_thread(root)
            if uids:
                found[folder.name] = uids
        if not found:
            return []
        if not self.sent & set(found):
            # Моего письма нет — нужна пометка. Сначала только флаги: тело чужой переписки качать незачем.
            flagged = False
            for name, uids in found.items():
                box.select(name)
                if any("\\Flagged" in f.flags for f in box.fetch(uids, body=False)):
                    flagged = True
                    break
            if not flagged:
                return []
        messages: List[MailMessage] = []
        by_id: Dict[str, MailMessage] = {}
        for name in sorted(found):
            box.select(name)
            for fetched in box.fetch(found[name]):
                m = self.message(fetched, name in self.sent)
                if m.message_id and m.message_id in by_id:  # копия письма в другой папке
                    by_id[m.message_id].mine |= m.mine
                    by_id[m.message_id].flagged |= m.flagged
                    continue
                if m.message_id:
                    by_id[m.message_id] = m
                messages.append(m)
        return messages

    def external_id(self, key: str) -> str:
        return f"root:{key}"
```
